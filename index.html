function sendMessage() { let input = document.getElementById("message"); let chat = document.getElementById("chat"); if (input.value.trim() === "") return; chat.innerHTML += `
${input.value}
`; chat.innerHTML += `
Thinking...
`; chat.scrollTop = chat.scrollHeight; setTimeout(() => { let bots = document.getElementsByClassName("bot"); bots[bots.length - 1].innerHTML = "I am BrightAI. Soon I will answer with real AI."; }, 800); input.value = ""; } document.getElementById("message").addEventListener("keypress", function(event) { if (event.key === "Enter") { sendMessage(); } });
